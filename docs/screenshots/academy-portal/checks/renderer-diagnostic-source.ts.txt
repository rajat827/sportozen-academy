import { test, expect } from "@playwright/test";
import { mkdir, writeFile } from "node:fs/promises";

// Temporary bounded diagnosis; this does not replace the customer acceptance flow.
test("single-page finance renderer diagnostic", async ({ playwright, browser }) => {
  test.setTimeout(90000);
  const origin = "http://127.0.0.1:3001";
  const engine = process.env.PLAYWRIGHT_CHANNEL || "bundled-chromium";
  const destination = `docs/screenshots/academy-portal/checks/${engine}-onboarding-renderer`;
  await mkdir(destination, { recursive: true });
  const api = await playwright.request.newContext({ baseURL: origin, extraHTTPHeaders: { Origin: origin } });
  const registration = await api.post("/api/auth/sign-up/email", { data: {
    name: "Synthetic Renderer Owner", email: `renderer-${crypto.randomUUID()}@example.test`, password: `Synthetic-only-${crypto.randomUUID()}`,
  } });
  expect(registration.status(), await registration.text()).toBe(200);
  const context = await browser.newContext({ storageState: await api.storageState(), viewport: { width: 1280, height: 720 } });
  const page = await context.newPage();
  const errors: string[] = [];
  page.on("pageerror", error => errors.push(error.message));
  const diagnostic: Record<string, unknown> = { engine };
  try {
    await page.goto("/workspace");
    await page.bringToFront();
    await page.getByLabel("Academy name", { exact: true }).fill("Synthetic Renderer Academy");
    await page.getByLabel("Academy contact email", { exact: true }).fill("renderer@example.test");
    await page.getByLabel("Country code", { exact: true }).fill("IN");
    await page.getByLabel("Currency code", { exact: true }).fill("INR");
    await page.getByLabel("Default branch time zone", { exact: true }).fill("Asia/Kolkata");
    await page.getByLabel("Date and number locale", { exact: true }).fill("en-IN");
    await page.getByRole("checkbox", { name: "fees", exact: true }).check();
    const created = page.waitForResponse(r => new URL(r.url()).pathname === "/api/workspaces" && r.request().method() === "POST");
    await page.getByRole("button", { name: "Create empty academy", exact: true }).click();
    const response = await created;
    expect(response.status(), await response.text()).toBe(201);
    const academy = await response.json();
    // Reproduce a previously active page becoming idle during real API setup.
    for (let i = 0; i < 10; i++) expect((await api.get(`/api/workspaces/${academy.id}`)).status()).toBe(200);
    const started = Date.now();
    const feeResponse = page.waitForResponse(r => new URL(r.url()).pathname === `/api/workspaces/${academy.id}/finance` && r.request().method() === "GET");
    await page.goto(`/workspace/${academy.id}?section=fees`);
    await page.bringToFront();
    const fee = await feeResponse;
    diagnostic.finance = { status: fee.status(), elapsedMs: Date.now() - started };
    const inspect = () => page.evaluate(() => {
      function box(selector: string) {
        const node = document.querySelector(selector);
        if (!node) return null;
        const rect = node.getBoundingClientRect(), style = getComputedStyle(node);
        return { text: node.textContent, rect: { x: rect.x, y: rect.y, width: rect.width, height: rect.height }, display: style.display, visibility: style.visibility, position: style.position, top: style.top, left: style.left, opacity: style.opacity, transform: style.transform, pointerEvents: style.pointerEvents, active: node === document.activeElement };
      }
      return { visibility: document.visibilityState, hidden: document.hidden, focus: document.hasFocus(), readyState: document.readyState, fonts: document.fonts.status, fontFaces: [...document.fonts].map(f => ({ family: f.family, status: f.status })), loading: document.body.textContent?.includes("Loading authorised fee records"), feePlans: [...document.querySelectorAll("button")].filter(n => n.textContent === "Fee Plans").length, skip: box(".skip-link"), tabs: box(".pf-tabs"), firstTab: box(".pf-tabs button"), h1Font: document.querySelector("h1") ? getComputedStyle(document.querySelector("h1")!).fontFamily : null };
    });
    diagnostic.afterResponse = await inspect();
    diagnostic.raf = await page.evaluate(() => Promise.race([
      new Promise(resolve => requestAnimationFrame(timestamp => resolve({ tick: true, timestamp }))),
      new Promise(resolve => setTimeout(() => resolve({ tick: false, elapsedMs: 3000 }), 3000)),
    ]));
    diagnostic.afterRaf = await inspect();
    await page.screenshot({ path: `${destination}/after-response.png`, timeout: 10000 });
    diagnostic.afterScreenshot = await inspect();
    try {
      await page.getByRole("button", { name: "Fee Plans", exact: true }).click({ timeout: 15000 });
      diagnostic.click = "passed";
    } catch (error) { diagnostic.click = error instanceof Error ? error.message : String(error); }
    diagnostic.afterClick = await inspect();
    await page.screenshot({ path: `${destination}/after-click.png`, timeout: 10000 });
    await page.setViewportSize({ width: 390, height: 844 });
    diagnostic.mobile = await inspect();
    await page.screenshot({ path: `${destination}/mobile.png`, timeout: 10000, fullPage: true });
    expect(diagnostic.click).toBe("passed");
  } finally {
    diagnostic.errors = errors;
    await writeFile(`${destination}/diagnostic.json`, JSON.stringify(diagnostic, null, 2));
    await context.close();
    await api.dispose();
  }
});
